Un login real: ¿qué guardamos en localStorage?

No solo el token de sesión — también el usuario completo, una preferencia y una fecha. Y al cerrar sesión, no todo tiene por qué borrarse.

async function login(email, password) {   const res = await fetch('/api/login', { method:'POST', body: ... });   const data = await res.json(); // { token, usuario }   localStorage.setItem('token', data.token);   localStorage.setItem('usuario', JSON.stringify(data.usuario));   localStorage.setItem('tema', 'oscuro'); // preferencia   localStorage.setItem('ultimoLogin', new Date().toISOString()); }   // ...al cargar la página (aunque se haya recargado)... const usuario = JSON.parse(localStorage.getItem('usuario'));   function logout() {   localStorage.removeItem('token');   localStorage.removeItem('usuario');   localStorage.removeItem('ultimoLogin');   // 'tema' se queda: es una preferencia, no algo de la sesión }
