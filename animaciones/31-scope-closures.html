Scope chain y closures

Una función interna ve su propio ámbito y el de todas las funciones que la envuelven, nunca al revés — y esa cadena de ámbitos sigue viva incluso cuando la función exterior ya ha terminado.

let usuario = 'Ana'; function externa() {   let rol = 'admin';   function interna() {     let accion = 'login';     console.log(accion, rol, usuario);   }   return interna; } const fn = externa(); fn();
